### Fix 2 — Step 24 (Cart Drawer), missing attribute on product page form
The cart drawer JS listens for `submit` on `[data-add-to-cart-form]`. Step 16's `card-product.liquid` gets this attribute added in Step 24, but `templates/product.liquid` (built in Step 17) does NOT — meaning the main product page's own Add to Cart button won't open the drawer unless you manually add `data-add-to-cart-form` to that form too when you reach Step 24.

### Note — cart.total_price is in cents
`cart.total_price` and similar Shopify money values are always in the store's smallest currency unit (cents for USD). The free-shipping math in Step 24 (`10000` = $100.00) is correct as written — just test it with a real cart total before trusting the display, since this is an easy place to be off by 100x.

  button.addEventListener('click', () => {

**JavaScript — File: `assets/animations.js`** (global, reusable for ALL sections)
```javascript
const SCROLL_ANIMATION_TRIGGER = 'scroll-trigger';
const SCROLL_ANIMATION_OFFSCREEN = 'scroll-trigger--offscreen';

function initializeScrollAnimations(rootEl = document) {
  const elements = Array.from(rootEl.querySelectorAll(`.${SCROLL_ANIMATION_TRIGGER}`));
  if (!elements.length) return;

  const observer = new IntersectionObserver((entries) => {
    entries.forEach((entry) => {
      if (entry.isIntersecting) {
        entry.target.classList.remove(SCROLL_ANIMATION_OFFSCREEN);
        observer.unobserve(entry.target);
      }
    });
  }, { rootMargin: '0px 0px -100px 0px' });

  elements.forEach(el => observer.observe(el));
}

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', () => initializeScrollAnimations());
} else {
  initializeScrollAnimations();
}

if (window.Shopify && window.Shopify.designMode) {
  document.addEventListener('shopify:section:load', (e) => initializeScrollAnimations(e.target));
}
```

## Step 21: Product variants deep-dive (teaching step)

**Key concept:** Variants are SKU-level product variations — different price/image/stock per option. Different from a plain image gallery (multiple photos of the *same* product).

**Example — Gold Ring, Sizes 5–9:**
- 1 product "Gold Signet Ring," 9 variants (one per size)
- Each variant: different SKU (GOLD-RING-5, GOLD-RING-6...), possibly different price, different stock, optionally a size-specific photo

**In Shopify Admin:**
1. Products → [product] → Variants section
2. "Add option" → Name: "Size" → add values 5, 5.5, 6...9 → Shopify auto-creates 9 variant rows
3. Per row: set SKU, price, stock quantity
4. Optional: upload variant-specific images (click row → Image → upload)

**In Liquid (already built in Step 17):** variant picker uses `product.options_with_values`; selecting an option updates `product.selected_variant`; add-to-cart submits that variant's ID.

**Why this matters:** a variant is a separate SKU with its own inventory. A gallery is just multiple photos of one product. The Layered Bracelet (next step) uses a gallery, not variants, because it's one fixed bundle.

**No code step — admin data entry + conceptual understanding.**
**Commit:** `git commit --allow-empty -m "admin: set up product variants (size/color options with SKUs and stock)"`

```javascript
function openCartDrawer() {
  const drawer = document.getElementById('cart-drawer');
  drawer.classList.add('active');
  drawer.hidden = false;
  document.body.style.overflow = 'hidden';
}

function closeCartDrawer() {
  const drawer = document.getElementById('cart-drawer');
  drawer.classList.remove('active');
  drawer.hidden = true;
  document.body.style.overflow = '';
}

document.querySelectorAll('[data-add-to-cart-form]').forEach(form => {
  form.addEventListener('submit', async (e) => {
    e.preventDefault();
    const formData = new FormData(form);
    const response = await fetch('/cart/add.js', { method: 'POST', body: formData });
    if (response.ok) {
      openCartDrawer();
      const cartResponse = await fetch('/cart.js');
      const cart = await cartResponse.json();
      updateCartDisplay(cart);
    }
  });
});

function updateCartDisplay(cart) {
  const badge = document.querySelector('[data-cart-count]');
  if (badge) badge.textContent = cart.item_count;
}

async function removeItem(key) {
  await fetch('/cart/change.js', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ line: key, quantity: 0 }) });
  location.reload();
}

async function updateQuantity(key, newQty) {
  if (newQty < 1) { await removeItem(key); return; }
  await fetch('/cart/change.js', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ line: key, quantity: newQty }) });
  location.reload();
}
```

class Wishlist {
  constructor() {

  }

## Step 30: Recently viewed products (optional)

**File: `assets/recently-viewed.js`**
```javascript
class RecentlyViewed {
  constructor() {
    this.storageKey = 'amara-recently-viewed';
    this.items = JSON.parse(localStorage.getItem(this.storageKey) || '[]');
  }
  add(productId) {
    this.items = this.items.filter(id => id !== productId);
    this.items.unshift(productId);
    if (this.items.length > 5) this.items.pop();
    this.save();
  }
  save() { localStorage.setItem(this.storageKey, JSON.stringify(this.items)); }
}
window.recentlyViewed = new RecentlyViewed();
if (window.Shopify && window.Shopify.product) { window.recentlyViewed.add(window.Shopify.product.id); }
```
